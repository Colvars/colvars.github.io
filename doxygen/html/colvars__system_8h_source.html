80 inline COLVARS_HOST_DEVICE void
81 cvm::system_boundary_conditions::set_boundaries(bool periodic_x_in, bool periodic_y_in,
82  bool periodic_z_in, cvm::rvector const &A,
83  cvm::rvector const &B, cvm::rvector const &C)
84 {
85  constexpr double diagonal_tol2 = 1.0e-10;
86 
87  periodic_x = periodic_x_in;
88  periodic_y = periodic_y_in;
89  periodic_z = periodic_z_in;
90 
91  // Avoid using stale reciprocal vectors when switching boundary types (especially for mixed periodicity)
92  reciprocal_cell_x.reset();
93  reciprocal_cell_y.reset();
94  reciprocal_cell_z.reset();
95  if ((periodic_x == periodic_y) && (periodic_x == periodic_z)) {
96  if (periodic_x) {
97  // Temporarily set as fully-periodic & orthogonal; will check below for triclinic
98  set_type(types::pbc_orthogonal);
99  } else {
100  reset();
101  return;
102  }
103  } else {
104  set_type(types::mixed);
105  }
106 
107  bool off_diagonal = false;
108 
109  if (periodic_x) {
110  unit_cell_x = A;
111  if ((A.y * A.y) > diagonal_tol2 || (A.z * A.z) > diagonal_tol2) {
112  off_diagonal = true;
113  } else {
114  reciprocal_cell_x = unit_cell_x/unit_cell_x.norm2();
115  }
116  } else {
117  unit_cell_x.reset();
118  reciprocal_cell_x.reset();
119  }
120 
121  if (periodic_y) {
122  unit_cell_y = B;
123  if ((B.x * B.x) > diagonal_tol2 || (B.z * B.z) > diagonal_tol2) {
124  off_diagonal = true;
125  } else {
126  reciprocal_cell_y = unit_cell_y/unit_cell_y.norm2();
127  }
128  } else {
129  unit_cell_y.reset();
130  reciprocal_cell_y.reset();
131  }
132 
133  if (periodic_z) {
134  unit_cell_z = C;
135  if ((C.x * C.x) > diagonal_tol2 || (C.y * C.y) > diagonal_tol2) {
136  off_diagonal = true;
137  } else {
138  reciprocal_cell_z = unit_cell_z/unit_cell_z.norm2();
139  }
140  } else {
141  unit_cell_z.reset();
142  reciprocal_cell_z.reset();
143  }
144 
145  if (type() == types::pbc_orthogonal && off_diagonal) {
146  set_type(types::pbc_triclinic);
147  }
148 
149  if (type() == types::pbc_triclinic) {
150  cvm::rvector const v_yz = cvm::rvector::outer(unit_cell_y, unit_cell_z);
151  reciprocal_cell_x = v_yz / (v_yz * unit_cell_x);
152  cvm::rvector const v_zx = cvm::rvector::outer(unit_cell_z, unit_cell_x);
153  reciprocal_cell_y = v_zx / (v_zx * unit_cell_y);
154  cvm::rvector const v_xy = cvm::rvector::outer(unit_cell_x, unit_cell_y);
155  reciprocal_cell_z = v_xy / (v_xy * unit_cell_z);
156  }
157 }

160 inline COLVARS_HOST_DEVICE
161 cvm::rvector cvm::system_boundary_conditions::position_distance(cvm::atom_pos const &pos1,
162  cvm::atom_pos const &pos2) const
163 {
164  cvm::rvector diff = (pos2 - pos1);
165 
166  if (type() == types::non_periodic || type() == types::unsupported) {
167  return diff;
168  }
169 
170  cvm::real const x_shift = ::floor(reciprocal_cell_x * diff + 0.5);
171  cvm::real const y_shift = ::floor(reciprocal_cell_y * diff + 0.5);
172  cvm::real const z_shift = ::floor(reciprocal_cell_z * diff + 0.5);
173 
174  diff.x -= x_shift * unit_cell_x.x + y_shift * unit_cell_y.x + z_shift * unit_cell_z.x;
175  diff.y -= x_shift * unit_cell_x.y + y_shift * unit_cell_y.y + z_shift * unit_cell_z.y;
176  diff.z -= x_shift * unit_cell_x.z + y_shift * unit_cell_y.z + z_shift * unit_cell_z.z;
177 
178  if (type() != types::pbc_orthogonal) {
179  // Matches both "mixed" and "pbc_triclinic", because reciprocal cell vectors are not used
180  diff += get_triclinic_shift(diff);
181  }
182 
183  return diff;
184 }

187 inline COLVARS_HOST_DEVICE cvm::rvector
188 cvm::system_boundary_conditions::get_triclinic_shift(cvm::rvector const &diff) const
189 {
190  cvm::real min_dist2 = diff.norm2();
191  cvm::rvector result{0.0, 0.0, 0.0};
192 
193  int const nx = periodic_x ? 1 : 0;
194  int const ny = periodic_y ? 1 : 0;
195  int const nz = periodic_z ? 1 : 0;
196 
197  // Loop over neighboring cells to find a shorter distance
198  for (int ix = -nx; ix <= nx; ix++) {
199  for (int iy = -ny; iy <= ny; iy++) {
200  for (int iz = -nz; iz <= nz; iz++) {
201  cvm::rvector const shift = ix * unit_cell_x + iy * unit_cell_y + iz * unit_cell_z;
202  cvm::real const this_dist2 = (diff + shift).norm2();
203  if (this_dist2 < min_dist2) {
204  result = shift;
205  min_dist2 = this_dist2;
206  }
207  }
208  }
209  }
210 
211  return result;
212 }
