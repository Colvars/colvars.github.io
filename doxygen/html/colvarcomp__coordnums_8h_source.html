185template <int flags, int static_en, int static_ed>
186inline COLVARS_HOST_DEVICE cvm::real colvar::coordnum::switching_function(
187 cvm::real const &l2, cvm::real &dFdl2,
188 int en, int ed,
189 cvm::real pairlist_tol)
190{
191 constexpr bool static_ed_is_2en = (static_en > 0) && (static_ed > 0) && (static_ed == 2 * static_en);
192 if constexpr ((static_en > 0) && (static_ed > 0)) {
193 // Assume en and ed are even integers, and avoid sqrt in the following
194 static_assert(static_en % 2 == 0, "static_en must be an even positive integer.");
195 static_assert(static_ed % 2 == 0, "static_ed must be an even positive integer.");
196 }
197 // Assume en and ed are even integers, and avoid sqrt in the following
198 cvm::real xn, xn_1, xd, en2_r, ed2_r;
199 if constexpr ((static_en > 0) && (static_ed > 0)) {
200 if constexpr (static_ed_is_2en) {
201 // xn = cvm::positive_integer_power<static_en / 2>(l2);
202 if constexpr (static_en == 2) {
203 xn_1 = 1.0;
204 } else {
205 xn_1 = cvm::positive_integer_power<static_en / 2 - 1>(l2);
206 }
207 xn = xn_1 * l2;
208 en2_r = (cvm::real)static_en / 2;
209 } else {
210 xn = cvm::positive_integer_power<static_en / 2>(l2);
211 xd = cvm::positive_integer_power<static_ed / 2>(l2);
212 en2_r = (cvm::real)static_en / 2;
213 ed2_r = (cvm::real)static_ed / 2;
214 }
215 } else {
216 int const en2 = en/2;
217 int const ed2 = ed/2;
218 xn = cvm::integer_power(l2, en2);
219 xd = cvm::integer_power(l2, ed2);
220 en2_r = (cvm::real) en2;
221 ed2_r = (cvm::real) ed2;
222 }
223
224 cvm::real const eps_l2 = 1.0e-7;
225 cvm::real const h = l2 - 1.0;
226 cvm::real func_no_pairlist;
227
228 if constexpr (static_ed_is_2en) {
229 func_no_pairlist = 1.0 / (1.0 + xn);
230 } else {
231 if (std::abs(h) < eps_l2) {
232 // Order-2 Taylor expansion: c0 + c1*h + c2*h^2
233 cvm::real const c0 = en2_r / ed2_r;
234 cvm::real const c1 = (en2_r * (en2_r - ed2_r)) / (2.0 * ed2_r);
235 cvm::real const c2 = (en2_r * (en2_r - ed2_r) * (2.0 * en2_r - ed2_r - 3.0)) / (12.0 * ed2_r);
236 func_no_pairlist = c0 + h * (c1 + h * c2);
237 } else {
238 func_no_pairlist = (1.0 - xn) / (1.0 - xd);
239 }
240 }
241
242 cvm::real func, inv_one_pairlist_tol;
243 if (flags & ef_use_pairlist) {
244 inv_one_pairlist_tol = 1 / (1.0-pairlist_tol);
245 func = (func_no_pairlist - pairlist_tol) * inv_one_pairlist_tol;
246 } else {
247 func = func_no_pairlist;
248 }
249
250 // If the value is too small and we are correcting for the tolerance, the result is negative
251 // and we need to exclude it rather than let it contribute to the sum or the gradients.
252 if (func < 0)
253 return 0;
254
255 if (flags & ef_gradients) {
256 if constexpr (static_ed_is_2en) {
257 const cvm::real deriv = -func_no_pairlist * func_no_pairlist * en2_r * xn_1;
258 dFdl2 = (flags & ef_use_pairlist) ? inv_one_pairlist_tol * deriv : deriv;
259 } else {
260 // Logarithmic derivative: 1st-order Taylor expansion around l2 = 1
261 cvm::real log_deriv;
262 if (std::abs(h) < eps_l2) {
263 cvm::real const g0 = 0.5 * (en2_r - ed2_r);
264 cvm::real const g1 = ((en2_r - ed2_r) * (en2_r + ed2_r - 6.0)) / 12.0;
265 log_deriv = g0 + h * g1;
266 } else {
267 log_deriv = (ed2_r * xd / ((1.0 - xd) * l2)) - (en2_r * xn / ((1.0 - xn) * l2));
268 }
269 dFdl2 = (flags & ef_use_pairlist) ?
270 func_no_pairlist * inv_one_pairlist_tol * log_deriv :
271 func * log_deriv;
272 }
273 }
274
275 return func;
276}

279template<int flags, int static_en, int static_ed>
280inline COLVARS_HOST_DEVICE cvm::real colvar::coordnum::compute_pair_coordnum(
281 cvm::rvector const &inv_r0_vec,
282 cvm::rvector const &inv_r0sq_vec,
283 int en,
284 int ed,
285 const cvm::real a1x,
286 const cvm::real a1y,
287 const cvm::real a1z,
288 const cvm::real a2x,
289 const cvm::real a2y,
290 const cvm::real a2z,
291 cvm::real& g1x,
292 cvm::real& g1y,
293 cvm::real& g1z,
294 cvm::real& g2x,
295 cvm::real& g2y,
296 cvm::real& g2z,
297 cvm::real pairlist_tol,
298 cvm::real pairlist_tol_l2_max,
299 cvm::system_boundary_conditions const &bc)
300{
301 const cvm::atom_pos pos1{a1x, a1y, a1z};
302 const cvm::atom_pos pos2{a2x, a2y, a2z};
303
304 cvm::rvector const diff = bc.position_distance(pos1, pos2);
305 cvm::rvector const scal_diff(diff.x * inv_r0_vec.x,
306 diff.y * inv_r0_vec.y,
307 diff.z * inv_r0_vec.z);
308 cvm::real const l2 = scal_diff.norm2();
309 if (flags & ef_use_pairlist) {
310 if (l2 > pairlist_tol_l2_max) {
311 // Exit if the distance is such that F(l2) < pairlist_tol
312 return 0.0;
313 }
314 }
315
316 cvm::real dFdl2 = 0.0;
317 cvm::real F = switching_function<flags, static_en, static_ed>(l2, dFdl2, en, ed, pairlist_tol);
318
319 if ((flags & ef_gradients) && (F > 0.0)) {
320 cvm::rvector const dl2dx((2.0 * inv_r0sq_vec.x) * diff.x,
321 (2.0 * inv_r0sq_vec.y) * diff.y,
322 (2.0 * inv_r0sq_vec.z) * diff.z);
323
324 const cvm::rvector G = dFdl2*dl2dx;
325 g1x += -1.0*G.x;
326 g1y += -1.0*G.y;
327 g1z += -1.0*G.z;
328 g2x += G.x;
329 g2y += G.y;
330 g2z += G.z;
331 }
332
333 return F;
334}
