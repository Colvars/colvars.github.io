120  static COLVARS_HOST_DEVICE inline real integer_power(real const &x, int const n)
121  {
122  // Original code: math_special.h in LAMMPS
123  double yy, ww;
124  if (x == 0.0) return 0.0;
125  int nn = (n > 0) ? n : -n;
126  ww = x;
127  for (yy = 1.0; nn != 0; nn >>= 1, ww *=ww) {
128  if (nn & 1) yy *= ww;
129  }
130  return (n > 0) ? yy : 1.0/yy;
131  }
